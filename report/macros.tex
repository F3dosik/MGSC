%%%%%%%%%%%% \image %%%%%%%%%%%%%%%%%%

% \image {Имя изображения.расширение}{Подпись к рисунку}{Скейл Изображения}

\newcommand{\image}[3]{
\begin{figure}[H]
	\centering
	\includegraphics[width=#3\textwidth]{#1}
	\caption{#2}
\end{figure}
}

%%%%%%%%%% \codefromfile  %%%%%%%%%%%%%%%%%%

% \codefromfile {Имя файла} {Язык listings, например Go или Python}

\newcommand{\codefromfile}[2]{
\begin{code}
	\lstinputlisting[language=#2]{Listings/#1}
\end{code}
}
